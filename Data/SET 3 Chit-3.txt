Generate Fibonacci numbers and determine the number of computational
steps required. 

import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter the number of terms: ");
        int n = sc.nextInt();

        int a = 0, b = 1;
        int steps = 0;

        System.out.println("Fibonacci Series:");

        for (int i = 0; i < n; i++) {
            System.out.print(a + " ");

            int c = a + b;
            a = b;
            b = c;

            steps++;
        }

        System.out.println("\nComputational Steps: " + steps);

        sc.close();
    }
}
